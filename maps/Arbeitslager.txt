from itertools import product
import os

from arenalib.maptools import WorldVXL, CTF, scandir_on_seed_generation, open_in_subdir

from horseradish.config import config

name        = "Arbeitslager"
team1_name  = "Officers"
team1_color = (255, 64, 0)
team2_name  = "Rebels"
team2_color = (64, 255, 0)
cap_limit   = 7

extensions = CTF(
    arena_blue_spawns      = [(122 + Δx, 256 + Δy, 0) for Δx, Δy in product(range(-32, 32), repeat = 2)],
    arena_green_spawns     = [(390 + Δx, 256 + Δy, 0) for Δx, Δy in product(range(-32, 32), repeat = 2)],
    arena_blue_base        = (128, 256, 0),
    arena_blue_flag        = (104, 256, 0),
    arena_green_base       = (384, 256, 0),
    arena_green_flag       = (408, 256, 0),
    arena_give_builder_kit = True,
    arena_respawn_time     = 10
)

on_seed_generation = scandir_on_seed_generation

def gen_script(basename, seed):
    savedir = os.path.join(config.config_dir, "saves")
    os.makedirs(savedir, exist_ok = True)

    try:
        script_file = open_in_subdir("saves", "{}.txt".format(seed))
    except FileNotFoundError:
        pass
    else:
        try:
            exec(compile(script_file.read(), script_file.name, 'exec'), globals())
        finally:
            script_file.close()

    filename = "{}.vxl".format(seed)
    return WorldVXL(os.path.join(savedir, filename))
